9) كشف العناصر الفريدة والمكررة في قائمة نصية التدريب رقم 3 - البرنامج الأول 
def find_duplicates(y):
    d = []        # قائمة حفظ العناصر الفريدة
    dd = []       # قائمة حفظ العناصر المكررة
    for i in y:
        if i in d and i not in dd:
            dd.append(i)
        if i not in d:
            d.append(i)
    return d, dd

# قائمة البيانات التوضيحية
name = ["ali", "sam", "omr", "sam", "omr", "sam", "asma", "tom"]

d, dd = find_duplicates(name)

print("العناصر الفريدة (Unique):", d)
print("العناصر المكررة (Duplicates):", dd)

# ---------- Output ----------
# العناصر الفريدة (Unique): ['ali', 'sam', 'omr', 'asma', 'tom']
# العناصر المكررة (Duplicates): ['sam', 'omr']

def find_missing_unsorted(n):
    num = set(n)
    out = []
    for i in range(min(n), max(n)):
        if i not in num:
            out.append(i)

    return out

list1 = [6, 7, 1, 3]
l1 = find_missing_unsorted(list1)
print("القيم المفقودة في القائمة غير المرتبة:", l1)

# ---------- Output ----------
# القيم المفقودة في القائمة غير المرتبة: [2, 4, 5]

from difflib import SequenceMatcher

# 1. إدخال النصوص المراد مقارنتها

text2 = "my name is amren"

# نسبة الاختلاف: 6.25%
20) حساب المسافة الإقليدية لتقييم تشابه المنتجات التدريب رقم 6 - البرنامج الأول (NLP) 
products = {"Laptop A": [1000, 4.5], "Laptop B": [950, 4.3], "Phone X": [300, 4.0]}
new_item = [980, 4.4]
distances = {}

for name, feat in products.items():
    squared_differences = []
    for p, n in zip(feat, new_item):
        diff = p - n          # الفرق بين الخاصيتين
        squared_diff = diff ** 2    # تربيع الفرق
        squared_differences.append(squared_diff)

    # جمع مربعات الفروق
    sum_of_squares = sum(squared_differences)

    # أخذ الجذر التربيعي للمجموع (حساب المسافة الإقليدية)
    euclidean_distance = sum_of_squares ** 0.5

    # تقريب الناتج وتخزينه في القاموس تحت اسم المنتج الحالي
    distances[name] = round(euclidean_distance, 2)

print("المسافات الحسابية:", distances)
print("المنتج الأقرب هو:", min(distances, key=distances.get))

# ---------- Output ----------
# المسافات الحسابية: {'Laptop A': 20.0, 'Laptop B': 30.0, 'Phone X': 680.0}
# المنتج الأقرب هو: Laptop A

22) القواعد الاقترانية (Association Rules) التدريب رقم 7 
import itertools
from collections import defaultdict

transactions = [
    ["جبن", "حليب", "خبز"],
    ["حليب", "خبز"],
    ["خبز", "تفاح"],
    ["جبن", "حليب"],
    ["جبن", "حليب", "خبز"]
]

total = len(transactions)
min_support = 0.4
min_confidence = 0.6

item_counts = defaultdict(int)
pair_counts = defaultdict(int)

for trans in transactions:
    for item in trans:
        item_counts[item] += 1

    pairs = itertools.combinations(sorted(trans), 2)
    for pair in pairs:
        pair_counts[pair] += 1

print("القواعد الاقترانية اكتشفت:")
for pair, count in pair_counts.items():
    support = count / total

    if support >= min_support:
        itemA = pair[0]
        itemB = pair[1]

        conf_A_B = count / item_counts[itemA]
        if conf_A_B >= min_confidence:
            print(f"إذا اشترى {itemA} -> يشتري {itemB} (ثقة: {conf_A_B:.0%})")

        conf_B_A = count / item_counts[itemB]
        if conf_B_A >= min_confidence:
            print(f"إذا اشترى {itemB} -> يشتري {itemA} (ثقة: {conf_B_A:.0%})")

# ---------- Output ----------
# القواعد الاقترانية اكتشفت:
# إذا اشترى جبن -> يشتري حليب (ثقة: 100%)
# إذا اشترى حليب -> يشتري جبن (ثقة: 75%)
# إذا اشترى جبن -> يشتري خبز (ثقة: 67%)
# إذا اشترى خبز -> يشتري حليب (ثقة: 75%)
# إذا اشترى حليب -> يشتري خبز (ثقة: 75%)
